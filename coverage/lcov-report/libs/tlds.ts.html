/**
 * Check whether a domain's TLD is valid.
 * Extracts the TLD from the hostname portion of a URL or bare domain.
 * Returns true if the TLD is in the IANA list.
 */
export function hasValidTLD(urlOrDomain: string): boolean {
  try {
    // Strip protocol and path to get hostname
    let host = urlOrDomain.replace(/^(?:https?:\/\/)?(?:www\.)?/i, "");
    // Remove path, query, hash
    host = host.split(/[/?#]/)[0];
    // Remove port
    host = host.split(":")[0];
    if (!host) return false;
    // Get TLD (last segment after final dot)
    const parts = host.split(".");
    if (parts.length < 2) return false;
    const tld = parts[parts.length - 1].toLowerCase();
    return TLD_SET.has(tld);
  } catch {
    return false;
  }
}
